# === 诊断 compact gather token selection ===

echo "=== output content ===" >> tmp.txt
grep -n "data:.*content" vllm_server.log | tail -20 >> tmp.txt

# 2. sel_diag 日志（前 10 条 + 最后 5 条）
echo "=== sel_diag (first 10) ===" >> tmp.txt
grep "sel_diag" vllm_server.log | head -10 >> tmp.txt
echo "=== sel_diag (last 5) ===" >> tmp.txt
grep "sel_diag" vllm_server.log | tail -5 >> tmp.txt
echo "=== sel_diag count ===" >> tmp.txt
grep -c "sel_diag" vllm_server.log >> tmp.txt

# 3. compact_gather 统计

grep -c "compact_gather.*enabled=1" vllm_server.log >> tmp.txt
